  async handleBoxTransition() {

      this.globalMonVar.productsInCurrentBox ===
      this.globalMonVar.limitProductInBox

  async deleteOldTablesIfNeeded(): Promise<void> {
    try {
      // Проверяем, нужно ли выполнять очистку сегодня
      const today = new Date().toISOString().split('T')[0];

      const lastCleanup = await this.entityManager.query(`
        SELECT last_cleanup_date FROM table_cleanup_metadata LIMIT 1
      `);

      if (
        lastCleanup[0]?.last_cleanup_date?.toISOString()?.split('T')[0] ===
        today
      ) {
        this.logger.info(
          'Очистка таблиц уже выполнялась сегодня___________________',
        );
        return;
      }

      // Выполняем очистку
      await this.deleteTablesOlderThan();

      // Обновляем дату последней очистки
      await this.entityManager.query(`
        UPDATE table_cleanup_metadata SET last_cleanup_date = CURRENT_DATE
      `);

      this.logger.info('Очистка таблиц выполнена успешно__________');
    } catch (err) {
      this.logger.error(
        'Ошибка при проверке необходимости очистки таблиц',
        err,
      );
      throw err;
    }
  }

  async deleteTablesOlderThan(): Promise<void> {
    this.logger.info(
      `Поиск таблиц старше ${isTableRetentionEnabled} дней______________`,
    );

    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - isTableRetentionEnabled);

    let deletedCount = 0;
    let errorCount = 0;

    try {
      const tables = await this.entityManager.query(`
        SELECT table_name
        FROM information_schema.tables
        WHERE table_schema = 'public'
          AND table_name ~ '^\\d'
      `);

      for (const table of tables) {
        const tableName = table.table_name;

        try {
          const dateMatch = tableName.match(/(\d{4})_(\d{2})_(\d{2})/);
          if (!dateMatch) continue;

          const year = parseInt(dateMatch[1], 10);
          const month = parseInt(dateMatch[2], 10) - 1;
          const day = parseInt(dateMatch[3], 10);

          // Валидация даты
          if (year < 2000 || year > new Date().getFullYear()) continue;
          const tableDate = new Date(year, month, day);
          if (isNaN(tableDate.getTime())) continue;

          if (tableDate < cutoffDate) {
            await this.entityManager.query(
              `DROP TABLE IF EXISTS "${tableName}" CASCADE`,
            );
            deletedCount++;
            await new Promise((resolve) => setTimeout(resolve, 100)); // Пауза
          }
        } catch (err) {
          errorCount++;
          this.logger.error(`Ошибка обработки таблицы ${tableName}`, err);
        }
      }

      this.logger.info(
        `Удаление завершено. Удалено таблиц: ${deletedCount}, ошибок: ${errorCount}`,
      );
    } catch (err) {
      this.logger.error('Ошибка в основном цикле удаления таблиц', err);
      throw err;
    }
  }

  async sendDBDataToFrontend(): Promise<void> {
    const startTime = performance.now();
    try {
      await this.checkTableExists(this.globalMonVar.tableName);

      // Один запрос вместо нескольких
      const query = `
        WITH
          product_count AS (SELECT COUNT(*) as product_count FROM "${this.globalMonVar.tableName}"),
          box_count AS (SELECT COUNT(DISTINCT box_number) as box_count FROM "${this.globalMonVar.tableName}"),
          pallet_count AS (SELECT COUNT(DISTINCT pallet_number) as pallet_count FROM "${this.globalMonVar.tableName}"),
          products_in_current_box AS (
            SELECT COUNT(*) as products_in_current_box
            FROM "${this.globalMonVar.tableName}"
            WHERE box_number = $1
          ),
          boxes_in_current_pallet AS (
            SELECT COUNT(DISTINCT box_number) as boxes_in_current_pallet
            FROM "${this.globalMonVar.tableName}"
            WHERE pallet_number = $2
          )
        SELECT
          pc.product_count,
          bc.box_count,
          plc.pallet_count,
          picb.products_in_current_box,
          bipc.boxes_in_current_pallet
        FROM
          product_count pc,
          box_count bc,
          pallet_count plc,
          products_in_current_box picb,
          boxes_in_current_pallet bipc
      `;

      const [result] = await this.entityManager.query(query, [
        this.globalMonVar.currentBoxNumber,
        this.globalMonVar.currentPalletNumber,
      ]);

      // Формируем объект allCounts напрямую из результата
      const allCounts: IAllCounts = {
        productCount: Number(result.product_count),
        productsInCurrentBox: Number(result.products_in_current_box),
        boxesInCurrentPallet: Number(result.boxes_in_current_pallet),
        boxCount: Number(result.box_count),
        palletCount: Number(result.pallet_count),
        currentBoxNumber: this.globalMonVar.currentBoxNumber,
        currentPalletNumber: this.globalMonVar.currentPalletNumber,
      };

      this.sendDataToFrontend(allCounts);
      this.updateGlobalVariablesFromCounts(allCounts);
      const executionTime = Math.round(performance.now() - startTime);
      console.log(
        `Оптимизированный метод sendDBDataToFrontend выполнен за ${executionTime} мс`,
      );
    } catch (err) {
      console.error('Ошибка в sendDBDataToFrontend:', err);
    }
  }

  async getLastBoxAndPalletInfo(): Promise<{
    lastBoxNumber: number;
    lastPalletNumber: number;
    productsInLastBox: number;
    boxesInLastPallet: number;
  }> {
    try {
      this.logger.info(
        'Код: Получение информации о последней коробке и паллете',
      );

      // Проверяем, есть ли открытая коробка
      const openBoxResult = await this.entityManager.query(
        `SELECT box_number, COUNT(*) as count
         FROM "${this.globalMonVar.tableName}"
         WHERE box_label IS NULL AND pallet_label IS NULL
         GROUP BY box_number
           LIMIT 1`,
      );

      let lastBoxNumber: number;
      let productsInLastBox: number;

      if (openBoxResult.length > 0) {
        // Есть открытая коробка - используем её
        lastBoxNumber = openBoxResult[0].box_number;
        productsInLastBox = parseInt(openBoxResult[0].count, 10);
      } else {
        // Нет открытых коробок - начинаем новую
        const maxBoxResult = await this.entityManager.query(
          `SELECT COALESCE(MAX(box_number), 0) as max_box FROM "${this.globalMonVar.tableName}"`,
        );
        lastBoxNumber = parseInt(maxBoxResult[0].max_box, 10) + 1;
        productsInLastBox = 0;
      }

      // Проверяем, есть ли открытая паллета
      const openPalletResult = await this.entityManager.query(
        `SELECT pallet_number, COUNT(DISTINCT box_number) as box_count
         FROM "${this.globalMonVar.tableName}"
         WHERE pallet_label IS NULL
         GROUP BY pallet_number
           LIMIT 1`,
      );

      let lastPalletNumber: number;
      let boxesInLastPallet: number;

      if (openPalletResult.length > 0) {
        // Есть открытая паллета - используем её
        lastPalletNumber = openPalletResult[0].pallet_number;
        boxesInLastPallet = parseInt(openPalletResult[0].box_count, 10);
      } else {
        // Нет открытых паллет - начинаем новую
        const maxPalletResult = await this.entityManager.query(
          `SELECT COALESCE(MAX(pallet_number), 0) as max_pallet FROM "${this.globalMonVar.tableName}"`,
        );
        lastPalletNumber = parseInt(maxPalletResult[0].max_pallet, 10) + 1;
        boxesInLastPallet = 0;
      }

      const lastBatchRecord = await this.batchHistoryService.getByBatchNumber(
        this.globalMonVar.task.batch,
        this.globalMonVar.task.date_manufacture,
      );

      if (!lastBatchRecord) {
        this.globalMonVar.currentPalletNumber = lastPalletNumber;
        this.globalMonVar.currentBoxNumber = lastBoxNumber;

        await this.batchHistoryService.saveOrUpdate(
          this.globalMonVar.task.batch,
          this.globalMonVar.currentBoxNumber,
          this.globalMonVar.currentPalletNumber,
          this.globalMonVar.task.date_manufacture,
        );
      } else {
        this.globalMonVar.currentBoxNumber = lastBatchRecord.last_box_number;
        this.globalMonVar.currentPalletNumber =
          lastBatchRecord.last_pallet_number;
      }

      this.globalMonVar.productsInCurrentBox = productsInLastBox;
      this.globalMonVar.boxesInCurrentPallet = boxesInLastPallet;

      this.logger.info(
        `Восстановлено: коробка ${lastBoxNumber} (продуктов: ${productsInLastBox}), паллета ${lastPalletNumber} (коробок: ${boxesInLastPallet})`,
      );

      return {
        lastBoxNumber,
        lastPalletNumber,
        productsInLastBox,
        boxesInLastPallet,
      };
    } catch (err) {
      console.log(err);
    }
  }

      const palletsMap: { [key: string]: IPallet } = {};

        if (!boxesMap[box_number]) {
          boxesMap[box_number] = { name: box_number, products: [] };

        boxesMap[box_number].products.push(code);

        if (!palletsMap[pallet_number]) {
          palletsMap[pallet_number] = { name: pallet_number, boxes: [] };
        }
        if (!palletsMap[pallet_number].boxes.includes(box_number)) {
          palletsMap[pallet_number].boxes.push(box_number);
        }
      });

      const boxes = Object.values(boxesMap).sort(
        (a, b) => parseInt(a.name) - parseInt(b.name),
      );
      const pallets = Object.values(palletsMap).sort(

      const boxCount = Object.keys(boxesMap).length;

      const palletCount = Object.keys(palletsMap).length;

      const lastBox = boxes[boxes.length - 1];

      // Номер последней паллеты
      const lastPallet = pallets[pallets.length - 1];

      // Количество продуктов в последней коробке
      const productsInLastBox = lastBox ? lastBox.products.length : 0;

      // Количество коробок в последней паллете
      const boxesInLastPallet = lastPallet ? lastPallet.boxes.length : 0;

  ParseCode(codes: string) {
    const partNumber = process.env.PART_NUMBER
      ? parseInt(process.env.PART_NUMBER, 10)
      : 3;

    let codesArr = codes.split('@').map((el) => el.replace(/^[\s\t\r\n]+/, ''));
    let barcod = '';
    let qrCod = '';

    console.log(
      'codesArrcodesArrcodesArrcodesArrcodesArrcodesArrcodesArrcodesArrcodesArrcodesArr',
      codesArr,
    );

    if (this.getApiStatus().isUsingApi && codesArr.length == 2) {
      return {
        barcod: '',
        qrCod: '',
        parse: false,
        message: 'Отсканировалось два кода при работе с апи',
      };
    }

    // Проверка на наличие двух частей
    if (!this.getApiStatus().isUsingApi && codesArr.length !== 2) {
      return {
        barcod,
        qrCod,
        parse: false,
        message: 'Ожидается 2 кода, разделенных символом "@"',
      };
    }

    if (
      !this.getApiStatus().isUsingApi &&
      (codesArr[0] === '' || codesArr[1] === '')
    ) {
      return {
        barcod,
        qrCod,
        parse: false,
        message: 'Один из кодов пуст',
      };
    }

    // Функция проверки на обычный код (начинается с 01 и содержит 21 на позициях 16-18)
    const isStandardCode = (code: string): boolean => {
      return code.startsWith('01') && code.slice(16, 18) === '21';
    };

    // Функция проверки на QR код (не является обычным кодом)
    const isQrCode = (code: string): boolean => {
      return !isStandardCode(code);
    };

    const codeForReturn = codesArr[0];

    if (!codeForReturn || codeForReturn == 'NoRead') {
      return {
        barcod: '',
        qrCod: '',
        parse: false,
        message: 'Ошибка при сканировании, камера ничего не передала',
      };
    }

    if (this.getApiStatus().isUsingApi && isStandardCode(codeForReturn)) {
      return {
        barcod: codeForReturn,
        qrCod: '',
        parse: true,
      };
    } else if (this.getApiStatus().isUsingApi && isQrCode(codeForReturn)) {
      return {
        barCod: '',
        qrCod: codeForReturn,
        parse: true,
      };
    }

    // Функция извлечения данных из QR

    // Сначала проверяем, какой из кодов является QR
    const code1IsQr = isQrCode(codesArr[0]);
    const code2IsQr = isQrCode(codesArr[1]);
    const code1IsStandard = isStandardCode(codesArr[0]);
    const code2IsStandard = isStandardCode(codesArr[1]);

    // Проверяем, что у нас есть ровно один QR код и ровно один обычный код
    if ((code1IsQr && code2IsQr) || (code1IsStandard && code2IsStandard)) {
      return {
        barcod,
        qrCod,
        parse: false,
        message:
          'Должен быть один QR код и один обычный код (начинающийся с 01 и содержащий 21 на 16-18 позициях)',
      };
    }

    let bruttoWeight = '';

    // Определяем, где QR код, а где обычный код
    if (code1IsQr && code2IsStandard) {
      const extracted = this.extractQrValue(codesArr[0]);
      qrCod = extracted.weight;
      bruttoWeight = extracted.brutto_weight;
      barcod = codesArr[1];
    } else if (code2IsQr && code1IsStandard) {
      const extracted = this.extractQrValue(codesArr[1]);
      qrCod = extracted.weight;
      bruttoWeight = extracted.brutto_weight || '';
      barcod = codesArr[0];
    } else {
      return {
        barcod,
        qrCod,
        parse: false,
        message: 'Не удалось определить тип кодов',
      };
    }

    // Дополнительная валидация извлеченных данных
    if (!qrCod || qrCod === '') {
      return {
        barcod,
        qrCod,
        parse: false,
        message: 'Не удалось извлечь данные из QR кода',
      };
    }

    if (!barcod || barcod === '') {
      return {
        barcod,
        qrCod,
        parse: false,
        message: 'Обычный код пуст',
      };
    }

    return {
      barcod,
      qrCod,
      brutto_weight: bruttoWeight,
      parse: true,
    };
  }

  extractQrValue = (qrString: string) => {
    const gs1Match = qrString.match(/\(3103\)(\d+)/);
    const gs2Match = qrString.match(/\(3303\)(\d+)/);

    if (gs1Match) {
      return {
        weight: gs1Match ? gs1Match[1] : undefined,
        brutto_weight: gs2Match ? gs2Match[1] : undefined,
      };
    }

    const parts = qrString.split('.');
    if (parts.length >= 2) {
      return {
        weight: parts[parts.length - 1],
        brutto_weight: parts[parts.length - 2],
      };
    }

    return { weight: qrString, brutto_weight: undefined };
  };

  simpleDetectCodeType(code) {
    if (!code || typeof code !== 'string') {
      return { type: 'unknown', value: '' };
    }

    const trimmed = code.trim();

    // Штрихкод: только цифры, длина от 8 до 14
    // Или специальный GS1 формат с 01...21...
    const isBarcode = code.startsWith('01') && code.slice(16, 18) === '21';
    const count = code.split(',').length;
    // QR код: всё остальное, что не подходит под штрихкод
    const isQR = !isBarcode && count > 0;

    return {
      type: isBarcode ? 'barcode' : isQR ? 'qr' : 'unknown',
      value: trimmed,
    };
  }

  async getWeightFromRemoteDb(code: string): Promise<string | null> {
    try {
      const column = this.remoteDbService.getRemoteWeightColumnName();
      const table = this.remoteDbService.getRemoteTableName();
      const codeColumn = this.remoteDbService.getRemoteCodeColumnName();

      const searchMode = process.env.CODE_SEARCH_MODE || '93';

      // Проверяем, что имена таблицы и колонки безопасны
      const isValidName = (name: string) =>
        /^[a-zA-Z_][a-zA-Z0-9_]*$/.test(name);

      if (!isValidName(table) || !isValidName(column)) {
        this.logger.error('Некорректное имя таблицы или колонки');
        return null;
      }

      // GS1 group separator (ASCII 29) — обозначает конец переменного поля
      const GS_SEPARATOR = '\x1D';

      const extractAfterDivider = (
        source: string,
        divider: string,
      ): string | null => {
        const pos = source.lastIndexOf(divider);
        if (pos === -1) return null;

        let value = source.substring(pos + divider.length);

        const gsPos = value.indexOf(GS_SEPARATOR);
        if (gsPos !== -1) {
          value = value.substring(0, gsPos);
        }

        return value;
      };

      let searchCode: string;

      if (searchMode === '21') {
        const extracted = extractAfterDivider(code, '21');
        if (extracted === null) {
          this.logger.warn(`Разделитель 21 не найден в коде ${code}`);
          return null;
        }
        searchCode = extracted;
      } else if (
        searchMode === '91' ||
        searchMode === '92' ||
        searchMode === '93'
      ) {
        const extracted = extractAfterDivider(code, searchMode);
        searchCode = extracted !== null ? extracted : code;
      } else {
        // Поведение по умолчанию: разделитель зависит от длины кода
        const divider = code.length > 60 ? '92' : '93';
        const extracted = extractAfterDivider(code, divider);
        searchCode = extracted !== null ? extracted : code;
      }

      this.logger.info(`Код для поиска: ${searchCode}`);

      // Используем LIKE для поиска кодов, начинающихся с searchCode
      const query = `SELECT [${column}] FROM [${table}] WHERE [${codeColumn}] LIKE @code`;

      let result = await this.remoteDbService.findOne(query, {
        code: searchCode + '%',
      });

      if (!result && this.is_using_own_db) {
        return '1.41';
      }

      if (result && result[column] !== undefined) {
        this.logger.info(`Получен вес для кода ${code}: ${result[column]}`);
        return result[column];
      }

  parseCodeForBezerba = (raw: string): { serial: string; payload: string } => {
    const code = raw.trim();
    const GS = '\u001d'; // GS1 group separator (ASCII 29) — управляющий символ
    const serialStart = 18;

    const gsPos = code.indexOf(GS, serialStart);

    const serial = code.substring(serialStart, gsPos);

    const payload = code.substring(gsPos + 1 + 2);

    return { serial, payload };
  };

  async uploadCodesForBezerba(): Promise<{
    fileName: string;
    filePath: string;
    processedCount: number;
  }> {
    const codes =
      this.globalMonVar.task.codes.length > 0
        ? this.globalMonVar.task.codes
        : [];

    this.logger.info(`Начало обработки ${codes.length} кодов для Bizerba`);

    const results: Array<{ serial: string; payload: string }> = [];

    for (const code of codes) {
      const parsed = this.parseCodeForBezerba(code);
      results.push({
        serial: parsed.serial,
        payload: parsed.payload,
      });
    }

    const fileName = 'UniqueData_TrueMark.txt';

    const targetDirectory = this.recordsCodesBezerbaDirectory;

    if (!fs.existsSync(targetDirectory)) {
      fs.mkdirSync(targetDirectory, { recursive: true });
      this.logger.info(`Создана директория: ${targetDirectory}`);
    }

    const filePath = join(targetDirectory, fileName);

    const header = 'GL6E|GT61|GT62|\n';
    const rows = results.map((result, index) => {
      const plu = (index + 1).toString();
      return `${plu}|${result.serial}|${result.payload}|\n`;
    });

    const fileContent = header + rows.join('');

    try {
      await fs.promises.writeFile(filePath, fileContent, 'utf-8');
      this.logger.info(`Файл успешно сохранён локально: ${filePath}`);

      try {
        await this.ftpService.uploadFile(
          fileContent,
          fileName,
          FtpDirectoryType.CODES,
        );
        this.logger.info(`Файл успешно загружен на FTP: ${fileName}`);
      } catch (ftpError) {
        this.logger.error(`Ошибка загрузки файла на FTP: ${ftpError.message}`);
      }

      return {
        fileName,
        filePath,
        processedCount: results.length,
      };
    } catch (error) {
      this.logger.error(`Ошибка сохранения файла: ${error.message}`);
      throw new Error(`Не удалось сохранить файл: ${error.message}`);
    }
  }

  determineCodeType(code: string): {
    type: SearchFrom | null;
    reason?: string;
  } {
    if (!code) {
      return {
        type: null,
        reason: 'Код не может быть пустым',
      };
    }

    if (code.length < 18) {
      return {
        type: null,
        reason: 'Код слишком короткий (меньше 18 символов)',
      };
    }

    const prefix = code.substring(0, 2);
    const after16 = code.length > 16 ? code.substring(16, 18) : '';

    // Проверка на код продукта
    if (prefix === '01' && after16 === '21') {
      return { type: SearchFrom.Code };
    }

    // Проверка на код коробки
    if (prefix === '01' && after16 === '11') {
      return { type: SearchFrom.Box };
    }

    // Проверка на код паллеты
    if (prefix === '02' && after16 === '11') {
      return { type: SearchFrom.Pallet };
    }

    console.log(
      "code.indexOf('21') != -1code.indexOf('21') != -1",
      code.indexOf('21') != -1,
    );

    if (
      prefix === '01' &&
      code.indexOf('21') != -1 &&
      code.indexOf('10') != -1 &&
      code.indexOf('11') != -1
    ) {
      return { type: SearchFrom.Box };
    }

    const cleanCode = code.replace(/[(]/g, '').replace(/[)]/g, '');

    if (cleanCode.startsWith('01')) {
      return { type: SearchFrom.Box };
    }

    if (cleanCode.startsWith('02')) {
      return { type: SearchFrom.Pallet };
    }

    // Определяем причину неизвестного формата
    let reason = 'Неизвестный формат кода';
    if (prefix !== '01' && prefix !== '02') {
      reason = `Неверный префикс (ожидался 01 или 02, получен ${prefix})`;
    } else if (after16 !== '21' && after16 !== '11') {
      reason = `Неверный идентификатор типа (ожидался 21 или 11, получен ${after16})`;
    }

    return {
      type: null,
      reason,
    };
  }

  async closeCurrentBox(code: string) {
    const tableName = this.getTableName();

    if (!tableName) {
      return {
        success: false,
        status: SearchStatus.ERROR,
        message: 'Таблица не найдена',
        data: null,
      };
    }

    try {
      const existingBoxLabel = await this.entityManager.query(
        `SELECT 1 FROM "${tableName}" WHERE box_label = $1`,
        [code],
      );

      if (existingBoxLabel.length > 0) {
        console.log('Этикетка коробки "${code}" уже существует в таблице');
        return {
          success: false,
          status: SearchStatus.ERROR,
          message: `Этикетка коробки "${code}" уже существует в таблице`,
          data: null,
        };
      }

      const codesToUpdate = await this.entityManager.query(
        `SELECT code FROM "${tableName}" WHERE box_label IS NULL AND pallet_label IS NULL `,
      );

      if (!codesToUpdate || codesToUpdate.length === 0) {
        return {
          success: false,
          status: SearchStatus.ERROR,
          message:
            'Нет доступных кодов для назначения коробки, нету свободных кодов  ',
          data: null,
        };
      }

      await this.entityManager.query(
        `UPDATE "${tableName}"
         SET box_label = $1
         WHERE box_label IS NULL AND pallet_label IS NULL`,
        [code],
      );

      this.logger.info(
        `Код коробки "${code}" успешно присвоен ${codesToUpdate.length} продуктам`,
      );

      await this.updateCountsForBox();
      await this.sendDBDataToFrontend();

      return {
        success: true,
        status: 'success',
        message: `Код коробки "${code}" успешно присвоен ${codesToUpdate.length} продуктам`,
        data: {
          boxLabel: code,
          updatedCodesCount: codesToUpdate.length,
          updatedCodes: codesToUpdate.map((row) => row.code),
        },
      };
    } catch (error) {
      this.logger.error(
        `Ошибка при сканировании кода коробки безербой: ${error.message}`,
        error,
      );
      return {
        success: false,
        status: SearchStatus.ERROR,
        message: `Ошибка при обработке кода коробки: ${error.message}`,
        data: null,
        error: error.message,
      };
    }
  }

        productCountInPallet,
      );

      return result;
    } catch (e) {

      return {
        success: false,
        status: SearchStatus.ERROR,
        message: 'Ошибка при закрытии и печати кода паллеты: ' + e.message,
        data: null,
        error: null,
      };
    }
  }

  extractBoxIdFromLabel(boxLabel: string): string | null {
    try {
      if (!boxLabel.includes('21')) {
        this.logger.warn(
          `Не удалось извлечь boxId из этикетки (отсутствует '21'): ${boxLabel}`,
        );
        return null;
      }

      if (boxLabel.length < 5) {
        this.logger.warn(
          `Строка этикетки слишком короткая для извлечения boxId: ${boxLabel}`,
        );
        return null;
      }

      const boxId = boxLabel.slice(-5);

      this.logger.info(`Извлечен boxId: ${boxId} из этикетки: ${boxLabel}`);
      return boxId;
    } catch (error) {
      this.logger.error(`Ошибка извлечения boxId: ${error.message}`);
      return null;
    }
  }

  async getBoxCodesCountFromRemoteDb(boxId: string): Promise<number> {
    this.logger.info(
      `Получение количества кодов для коробки ${boxId} из удаленной БД`,
    );

    try {
      const codes: string[] =
        await this.remoteDbService.getBoxCodesCountFromRemoteDb(boxId);

      const matched: { originalCode: string; fullCode: string }[] = [];

      const notMatchedCodes: string[] = [];

      for (const code of codes) {
        const processedRecord =
          await this.processedCodeRepository.findByCodeAfter21(code);

        if (!processedRecord) {
          notMatchedCodes.push(code);
          this.logger.warn(
            `Код ${code} из удаленной БД не сопоставлен с processed_codes`,
          );
          continue;
        }

        matched.push({
          originalCode: code,
          fullCode: processedRecord.fullCode,
        });
      }

      console.log('matchedmatchedmatchedmatchedmatched', matched);

      const taskCodes = this.globalMonVar.task.codes || [];
      let resultCodes = matched;

      if (taskCodes.length > 0) {
        const taskCodesSet = new Set(taskCodes.map((c) => c.trim()));
        resultCodes = matched.filter((c) =>
          taskCodesSet.has(c.fullCode.trim()),
        );

        const removed = matched.length - resultCodes.length;
        if (removed > 0) {
          this.logger.warn(`Не из задания: ${removed} кодов`);
        }
      }

      console.log('countcountcount from db', resultCodes.length);

      return resultCodes.length;
    } catch (error) {
      this.logger.error(
        `Ошибка при получении количества кодов коробки из удаленной БД: ${error.message}`,
      );
      throw error;
    }
  }

  async saveAllCodesToProcessedTable(codes: string[]): Promise<void> {
    if (!codes || codes.length === 0) {
      this.logger.warn(
        'Список кодов пуст, пропускаем сохранение в processed_codes',
      );
      return;
    }

    this.logger.info(
      `Начинаем сохранение ${codes.length} кодов в processed_codes`,
    );

    let savedCount = 0;
    let skippedCount = 0;
    let errorCount = 0;

    for (const code of codes) {
      try {
        const codeAfter21 = this.extractCodeAfter21(code);

        if (codeAfter21) {
          // Сохраняем в базу
          await this.processedCodeRepository.saveCode(code, codeAfter21);
          savedCount++;
        } else {
          this.logger.warn(
            `Не удалось извлечь часть после 21 для кода: ${code}`,
          );
          skippedCount++;
        }
      } catch (error) {
        errorCount++;
        this.logger.error(
          `Ошибка сохранения кода ${code} в processed_codes: ${error.message}`,
        );
      }
    }

    this.logger.info(
      `Сохранение кодов в processed_codes завершено. Сохранено: ${savedCount}, пропущено: ${skippedCount}, ошибок: ${errorCount}`,
    );
  }

  extractCodeAfter21(fullCode: string): string | null {
    try {
      if (!fullCode) return null;

      const code = fullCode.trim();
      const GS = '\x1D';
      const serialStart = 18; // 01 + 14 цифр GTIN + 21 = 18 символов

      // Проверяем структуру: 01<GTIN14>21<serial>
      if (
        code.length <= serialStart ||
        !code.startsWith('01') ||
        code.slice(16, 18) !== '21'
      ) {
        this.logger.warn(
          `Код не соответствует формату 01...21...: ${fullCode}`,
        );
        return null;
      }

      // Ищем GS только начиная с позиции серийника
      const gsPos = code.indexOf(GS, serialStart);

      // Если GS нет, серийник идёт до конца строки
      const serial = (
        gsPos === -1
          ? code.substring(serialStart)
          : code.substring(serialStart, gsPos)
      ).trim();

      if (!serial) {
        this.logger.warn(`Пустой серийник в коде: ${fullCode}`);
        return null;
      }

      this.logger.info(`Извлечён серийник: ${fullCode} -> ${serial}`);
      return serial;
    } catch (error) {
      this.logger.error(`Ошибка извлечения серийника: ${error.message}`);
      return null;
    }
  }

  async getCommonWeightOfCurrentPacking() {
    const { tableName, success, message } = await this.checkDataForTable();

    if (!success) {
      return {
        success: false,
        status: SearchStatus.ERROR,
        message: message,
        data: null,
        error: null,
      };
    }

    try {
      const result = await this.entityManager.query(
        `SELECT SUM(CAST(weight AS DECIMAL)) as total_weight FROM "${tableName}"`,
      );

      const totalWeight = result[0]?.total_weight || 0;

      return {
        success: true,
        status: SearchStatus.SUCCESS,
        message: 'Вес фасовки получен',
        data: {
          weight: parseFloat(totalWeight) || 0,
          unit: 'kg',
        },
      };

      console.error('Ошибка при получении веса фасовки:', e);
      return {
        success: false,
        status: SearchStatus.ERROR,
        message: 'Ошибка при получении веса фасовки',
        data: null,
        error: e.message,
      };
    }
  }
